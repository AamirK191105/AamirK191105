(async function () {
const username = "AamirK191105";

return;
}
const GH_HEADERS = {
"Accept": "application/vnd.github+json"
};
const CACHE_KEY = "sn-portfolio-gh-cache-v1";
const CACHE_TTL_MS = 30 * 60 * 1000;
const EXCLUDED_REPOS = new Set(["AamirK191105.github.io", "AamirK191105"]);
const isVisibleRepo = (repo) => repo && repo.name && !EXCLUDED_REPOS.has(repo.name);
const fetchWithRetry = async (url, options = {}, maxRetries = 3) => {
let lastError;
for (let attempt = 0; attempt < maxRetries; attempt++) {
try {
const response = await fetch(url, options);
if (response.ok) {
return response;
}
if (response.status === 404) {
throw new Error(`Not found: ${url}`);
}
if (response.status === 403) {
throw new Error("GitHub API rate limited");
}
lastError = new Error(`HTTP ${response.status}`);
} catch (error) {
lastError = error;
if (attempt < maxRetries - 1) {
const delay = Math.pow(2, attempt) * 1000;
await new Promise((resolve) => setTimeout(resolve, delay));
}
}
}
throw lastError;
};
const scrollToTop = (event) => {
if (event) {
event.preventDefault();
}
window.scrollTo({
top: 0,
behavior: "smooth"
});
};
document.querySelectorAll('a[href="#top"]').forEach((link) => {
link.addEventListener("click", scrollToTop);
});
const readCache = () => {
try {
const raw = localStorage.getItem(CACHE_KEY);
if (!raw) {
return null;
}
const parsed = JSON.parse(raw);
if (!parsed.timestamp || !parsed.data) {
return null;
}
if (Date.now() - parsed.timestamp > CACHE_TTL_MS) {
return null;
}
return parsed.data;
} catch {
return null;
}
};
const writeCache = (data) => {
try {
localStorage.setItem(CACHE_KEY, JSON.stringify({
timestamp: Date.now(),
data
}));
} catch {
// Ignore storage failures and continue with in-memory data.
}
};
const renderRepoCard = (repo) => {
const article = document.createElement("article");
article.className = "repo-item";
const h3 = document.createElement("h3");
const link = document.createElement("a");
link.href = repo.html_url;
link.target = "_blank";
link.rel = "noopener noreferrer";
link.textContent = repo.name;
h3.appendChild(link);
const about = document.createElement("p");
about.className = "repo-about";
about.textContent = repo.description || "No public description added yet.";
const topics = document.createElement("ul");
topics.className = "chip-list repo-topic-chips";
(repo.topics && repo.topics.length ? repo.topics : ["No Topics"]).forEach((topic) => {
const chip = document.createElement("li");
chip.textContent = topic;
topics.appendChild(chip);
});
article.appendChild(h3);
article.appendChild(about);
article.appendChild(topics);
return article;
};
const aggregateThisYearContributions = (events) => {
const year = new Date().getFullYear();
const contributionTypes = new Set([
"PullRequestEvent",
"IssuesEvent",
"IssueCommentEvent",
"PullRequestReviewCommentEvent",
"CommitCommentEvent",
"CreateEvent"
]);
return events
.filter((event) => event && event.created_at)
.filter((event) => new Date(event.created_at).getFullYear() === year)
.reduce((total, event) => {
if (event.type === "PushEvent") {
return total + (Number(event.payload && event.payload.size) || 1);
}
return total + (contributionTypes.has(event.type) ? 1 : 0);
}, 0);
};
const setupTrackNavigation = (trackElement, backControl, nextControl, cardSelector) => {
const updateButtons = () => {
const maxLeft = trackElement.scrollWidth - trackElement.clientWidth;
backControl.disabled = trackElement.scrollLeft <= 2;
nextControl.disabled = trackElement.scrollLeft >= maxLeft - 2;
};
const step = () => {
const firstCard = trackElement.querySelector(cardSelector);
if (!firstCard) {
return trackElement.clientWidth;
}
const styles = window.getComputedStyle(trackElement);
const gap = parseFloat(styles.columnGap || styles.gap || "0");
return (firstCard.getBoundingClientRect().width + gap) * 3;
};
backControl.addEventListener("click", () => {
trackElement.scrollBy({ left: -step(), behavior: "smooth" });
});
nextControl.addEventListener("click", () => {
trackElement.scrollBy({ left: step(), behavior: "smooth" });
});
trackElement.addEventListener("scroll", updateButtons, { passive: true });
window.addEventListener("resize", updateButtons);
return { updateButtons };
};
const repoNavigation = setupTrackNavigation(track, backButton, nextButton, ".repo-item");
try {
const cached = readCache();
let user;
let repos;
let events;
if (cached) {
user = cached.user;
repos = cached.repos;
events = cached.events;
} else {
const [userResponse, reposResponse, eventsResponse] = await Promise.all([
fetchWithRetry(`https://api.github.com/users/${username}`, { headers: GH_HEADERS }),
fetchWithRetry(`https://api.github.com/users/${username}/repos?per_page=100&sort=updated`, { headers: GH_HEADERS }),
fetchWithRetry(`https://api.github.com/users/${username}/events/public?per_page=100`, { headers: GH_HEADERS })
]);
if (!userResponse.ok || !reposResponse.ok || !eventsResponse.ok) {
throw new Error("Failed to fetch GitHub profile data.");
}
user = await userResponse.json();
repos = await reposResponse.json();
events = await eventsResponse.json();
writeCache({ user, repos, events });
}
const visibleRepos = Array.isArray(repos) ? repos.filter(isVisibleRepo) : [];
if (statRepos) {
statRepos.textContent = String(visibleRepos.length);
}
if (statSocial) {
statSocial.textContent = `${user.followers ?? 0} / ${user.following ?? 0}`;
}
if (statContrib) {
statContrib.textContent = String(aggregateThisYearContributions(Array.isArray(events) ? events : []));
}
track.innerHTML = "";
visibleRepos.forEach((repo) => {
track.appendChild(renderRepoCard(repo));
});
const uniqueTopics = Array.from(new Set(
visibleRepos.flatMap((repo) => Array.isArray(repo.topics) ? repo.topics : [])
)).sort((a, b) => a.localeCompare(b));
topicsContainer.innerHTML = "";
if (uniqueTopics.length === 0) {
const empty = document.createElement("span");
empty.className = "topic-chip";
empty.textContent = "No public topics found.";
topicsContainer.appendChild(empty);
} else {
uniqueTopics.forEach((topic) => {
const chip = document.createElement("span");
chip.className = "topic-chip";
chip.textContent = topic;
topicsContainer.appendChild(chip);
});
}
} catch {
const cached = readCache();
if (cached && Array.isArray(cached.repos) && cached.repos.length) {
if (statContrib) {
statContrib.textContent = String(aggregateThisYearContributions(Array.isArray(cached.events) ? cached.events : []));
}
track.innerHTML = "";
cached.repos.filter(isVisibleRepo).forEach((repo) => track.appendChild(renderRepoCard(repo)));
const fallbackTopics = Array.from(new Set(
cached.repos.filter(isVisibleRepo).flatMap((repo) => Array.isArray(repo.topics) ? repo.topics : [])
)).sort((a, b) => a.localeCompare(b));
topicsContainer.innerHTML = "";
if (fallbackTopics.length === 0) {
topicsContainer.innerHTML = "<span class=\"topic-chip\">No public topics found.</span>";
} else {
fallbackTopics.forEach((topic) => {
const chip = document.createElement("span");
chip.className = "topic-chip";
chip.textContent = topic;
topicsContainer.appendChild(chip);
});
}
} else {
if (statContrib) {
statContrib.textContent = "--";
}
track.innerHTML = "<article class=\"repo-item\"><h3>Could not load repositories</h3><p class=\"repo-about\">GitHub API limit reached. Please refresh later.</p><p class=\"repo-topics\">Topics: Unavailable</p></article>";
topicsContainer.innerHTML = "<span class=\"topic-chip\">Topics unavailable</span>";
}
}
repoNavigation.updateButtons();
})();
